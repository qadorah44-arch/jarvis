/* ---------- Orb animation ---------- */
const cv = document.getElementById('orb'), ctx = cv.getContext('2d');
let W, H, DPR = Math.min(devicePixelRatio || 1, 2);
let mode = 'idle', energy = 0, target = 0.15, t = 0;
function resize(){ W = cv.clientWidth; H = cv.clientHeight; cv.width = W*DPR; cv.height = H*DPR; ctx.setTransform(DPR,0,0,DPR,0,0); }
addEventListener('resize', resize); resize();
const parts = Array.from({length:180}, () => ({a:Math.random()*Math.PI*2, r:.35+Math.random()*.75, s:(Math.random()*.6+.2)*(Math.random()<.5?-1:1), z:Math.random()}));
function draw(){
  t += 0.016;
  target = mode==='speaking' ? 0.55 + Math.sin(t*9)*0.25 + Math.random()*0.2
         : mode==='thinking' ? 0.35 + Math.sin(t*4)*0.1
         : mode==='listening' ? 0.45 + Math.sin(t*6)*0.15
         : 0.15 + Math.sin(t*1.5)*0.04;
  energy += (target - energy) * 0.12;
  ctx.clearRect(0,0,W,H);
  const cx = W/2, cy = H/2, R = Math.min(W,H)*0.32;
  // glow
  const g = ctx.createRadialGradient(cx,cy,0,cx,cy,R*1.6);
  g.addColorStop(0, `rgba(120,230,255,${0.55+energy*0.4})`);
  g.addColorStop(0.15, `rgba(63,208,255,${0.35+energy*0.3})`);
  g.addColorStop(0.5, 'rgba(10,132,255,0.10)');
  g.addColorStop(1, 'rgba(0,0,0,0)');
  ctx.fillStyle = g; ctx.beginPath(); ctx.arc(cx,cy,R*1.6,0,7); ctx.fill();
  // rings
  ctx.lineWidth = 1;
  for(let i=0;i<5;i++){
    const rr = R*(0.55+i*0.14) * (1+energy*0.08*Math.sin(t*3+i));
    ctx.strokeStyle = `rgba(63,208,255,${0.10+0.06*(5-i)*(0.5+energy)})`;
    ctx.setLineDash([2+i*3, 6+i*4]);
    ctx.beginPath(); ctx.arc(cx,cy,rr, t*(i%2?-0.4:0.3)*(1+energy), t*(i%2?-0.4:0.3)*(1+energy)+Math.PI*2); ctx.stroke();
  }
  ctx.setLineDash([]);
  // particles
  for(const p of parts){
    p.a += p.s*0.01*(1+energy*3);
    const rr = R*p.r*(1+energy*0.25*Math.sin(t*5+p.a*3));
    const x = cx+Math.cos(p.a)*rr, y = cy+Math.sin(p.a)*rr*0.98;
    ctx.fillStyle = `rgba(140,225,255,${0.25+p.z*0.6})`;
    ctx.fillRect(x,y,1.4+p.z,1.4+p.z);
  }
  // core
  const core = R*(0.13+energy*0.07);
  const cg = ctx.createRadialGradient(cx,cy,0,cx,cy,core*2.2);
  cg.addColorStop(0,'#ffffff'); cg.addColorStop(0.35,'#9eeeff'); cg.addColorStop(1,'rgba(63,208,255,0)');
  ctx.fillStyle = cg; ctx.beginPath(); ctx.arc(cx,cy,core*2.2,0,7); ctx.fill();
  requestAnimationFrame(draw);
}

async function gemTTS(text){
  const models = gemModel ? [gemModel] : GEM_MODELS;
  let lastErr = '';
  for (const m of models) {
    const r = await fetch('https://generativelanguage.googleapis.com/v1beta/models/' + m + ':generateContent', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json', 'x-goog-api-key': cfg.gemKey },

        contents: [{ parts: [{ text }] }],
        generationConfig: { responseModalities: ['AUDIO'], speechConfig: { voiceConfig: { prebuiltVoiceConfig: { voiceName: cfg.gemVoice } } } }
      })
    });
    const d = await r.json().catch(() => ({}));
    const part = d.candidates?.[0]?.content?.parts?.find(p => p.inlineData);

  });
}

  return w.filter(x => mine.has(x)).length / w.length >= 0.5;
}
const STOP = /^(توقف|وقف|اسكت|أسكت|اصمت|كفى|كفايه|كفاية|خلاص|بس|انتظر|stop|wait|enough|shut up|hold on)/i;
function interrupt(said){
  stopSpeaking(); earsPaused = false; awakeUntil = Date.now() + 12000; _setMode('listening');
  if (STOP.test(said.trim())) { heard = ''; input.value = ''; return true; }   // just "stop"
  return false;
}

if (SR) {
  rec = new SR();
  rec.continuous = true; rec.interimResults = true; rec.lang = cfg.lang;
  rec.onresult = e => {
    if (busy) return;
    let fin = '', interim = '';
    for (let i = e.resultIndex; i < e.results.length; i++) {
      const t = e.results[i][0].transcript;
      if (e.results[i].isFinal) fin += t; else interim += t;
    }
    if (earsPaused) {                       // JARVIS is talking
      const said = (fin || interim).trim();
      if (!said || isEcho(said)) return;      // his own voice → ignore
      if (!STOP.test(said) && norm(said).length < 2 && !fin) return; // wait for a clearer word
      if (interrupt(said)) return;            // "stop" → silence, keep listening
    }
    if (fin) heard = (heard + ' ' + fin).trim();
    if (awake()) input.value = (heard + ' ' + interim).trim();
    clearTimeout(silence); silence = setTimeout(flush, fin && !interim ? 1100 : 1700);
  };
  rec.onend = () => { recActive = false; if (live) setTimeout(startRec, 250); };
  rec.onerror = e => {
    if (e.error === 'not-allowed' || e.error === 'service-not-allowed') { live = false; hint(); note.textContent = 'اسمح بالميكروفون من إعدادات المتصفح.'; setMode('idle'); }
  };
}
let greeted = false;
async function toggleLive(){
  unlock();
  if (!SR) { hint(); return; }
  if (!activeKey()) { openSettings(); return; }
  if (live && earsPaused) { interrupt('stop'); return; }   // tap while he talks = interrupt
  if (live) { live = false; try { rec.abort(); } catch(e){} recActive = false; hint(); setMode('idle'); return; }
  live = true; hint(); startRec(); _setMode('listening');
  if (!greeted) { greeted = true;
    const g = cfg.lang.startsWith('ar') ? `أهلاً سيدي، أنا ${cfg.name}. كل الأنظمة تعمل، كيف أساعدك؟` : `Good to see you, sir. ${cfg.name} online. How can I help?`;
    bubble('ai', g);
    pauseEars(); _setMode('speaking'); await say(g); awakeUntil = Date.now() + 12000; setMode('idle'); resumeEars();
  }
}
